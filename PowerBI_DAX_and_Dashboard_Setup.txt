-- Create these measures in Power BI after importing powerbi_superstore_data.csv

Total Sales = SUM('powerbi_superstore_data'[Sales])

Total Profit = SUM('powerbi_superstore_data'[Profit])

Profit Margin % = DIVIDE([Total Profit], [Total Sales], 0)

Total Orders = DISTINCTCOUNT('powerbi_superstore_data'[Order ID])

Average Order Value = DIVIDE([Total Sales], [Total Orders], 0)

Total Quantity = SUM('powerbi_superstore_data'[Quantity])

Average Discount % = AVERAGE('powerbi_superstore_data'[Discount])

Loss Making Lines =
CALCULATE(
    COUNTROWS('powerbi_superstore_data'),
    'powerbi_superstore_data'[Profit] < 0
)

Sales 2016 =
CALCULATE(
    [Total Sales],
    'powerbi_superstore_data'[Order Year] = 2016
)

Sales 2017 =
CALCULATE(
    [Total Sales],
    'powerbi_superstore_data'[Order Year] = 2017
)

Sales Growth 2017 vs 2016 % =
DIVIDE([Sales 2017] - [Sales 2016], [Sales 2016], 0)

Profit per Order =
DIVIDE([Total Profit], [Total Orders], 0)
